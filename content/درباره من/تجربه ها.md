  // Reveal on scroll
  var revealTargets = (root.querySelectorAll ? root.querySelectorAll('.xp-reveal') : document.querySelectorAll('.xp-reveal'));
  var timelineItems = (root.querySelectorAll ? root.querySelectorAll('.xp-timeline-item') : document.querySelectorAll('.xp-timeline-item'));

  if('IntersectionObserver' in window){
    var revealObserver = new IntersectionObserver(function(entries){
      entries.forEach(function(entry){
        if(entry.isIntersecting){
          entry.target.classList.add('xp-in');
          revealObserver.unobserve(entry.target);
        }
      });
    }, { threshold: 0.15, rootMargin: '0px 0px -60px 0px' });
    revealTargets.forEach(function(el){ revealObserver.observe(el); });

    var activeYearEl = document.getElementById('xpActiveYear');
    var activeObserver = new IntersectionObserver(function(entries){
      entries.forEach(function(entry){
        var item = entry.target;
        if(entry.isIntersecting){
          item.classList.add('xp-in');
        }
        if(entry.isIntersecting && entry.intersectionRatio > 0.5){
          timelineItems.forEach(function(t){ t.classList.remove('xp-active'); });
          item.classList.add('xp-active');
          var lvl = item.querySelector('.xp-level');
          if(lvl && activeYearEl){
            var yr = lvl.textContent.replace('+','');
            activeYearEl.innerHTML = 'Level <span>' + esc(yr) + '</span>';
          }
        }
      });
    }, { threshold: [0, 0.5, 1] });
    timelineItems.forEach(function(el){ activeObserver.observe(el); });
  } else {
    revealTargets.forEach(function(el){ el.classList.add('xp-in'); });
    timelineItems.forEach(function(el){ el.classList.add('xp-in'); });
  }

    var btn = e.target.closest ? e.target.closest('.xp-expand-btn') : null;
    if(!btn || !root.contains || !root.contains(btn)) {
      if(!btn) return;
    }
    var card = btn.closest('.xp-card');
    if(!card) return;
    var open = card.getAttribute('data-open') === 'true';
    card.setAttribute('data-open', String(!open));
    btn.setAttribute('aria-expanded', String(!open));
    btn.querySelector('span').textContent = open ? 'Read the full record' : 'Collapse';
  });

  // Timeline fill on scroll (single rAF-throttled listener)
  var fillEl = document.getElementById('xpFill');
  var timelineEl = document.getElementById('xpTimeline');
  var ticking = false;

  function updateFill(){
    ticking = false;
    if(!timelineEl || !fillEl) return;
    var rect = timelineEl.getBoundingClientRect();
    var vh = window.innerHeight || document.documentElement.clientHeight;
    var total = rect.height;
    var visibleTop = Math.min(Math.max(vh * 0.5 - rect.top, 0), total);
    var pct = total > 0 ? (visibleTop / total) * 100 : 0;
    fillEl.style.height = pct + '%';
  }

  function onScroll(){
    if(!ticking){
      ticking = true;
      window.requestAnimationFrame(updateFill);
    }
  }

  window.addEventListener('scroll', onScroll, { passive: true });
  window.addEventListener('resize', onScroll, { passive: true });
  updateFill();

})();
