  <script>
    (function () {
      'use strict';

      document.addEventListener('DOMContentLoaded', function () {
        const timelineWrapper = document.querySelector('.timeline-experience-wrapper');
        const items = timelineWrapper.querySelectorAll('.timeline-item');
        const progressLine = document.getElementById('timelineProgress');

        /* ------------------------------------------------------------------
           1. IntersectionObserver for Reveal Animations
           ------------------------------------------------------------------ */
        const observerOptions = {
          root: null,
          rootMargin: '0px 0px -10% 0px',
          threshold: 0.15
        };

        const itemObserver = new IntersectionObserver((entries) => {
          entries.forEach(entry => {
            if (entry.isIntersecting) {
              entry.target.classList.add('is-visible');
            }
          });
        }, observerOptions);

        items.forEach(item => itemObserver.observe(item));

        /* ------------------------------------------------------------------
           2. Smooth Scroll-Linked Progress Line Calculation
           ------------------------------------------------------------------ */
        function updateTimelineProgress() {
          if (!timelineWrapper || !progressLine) return;

          const wrapperRect = timelineWrapper.getBoundingClientRect();
          const windowHeight = window.innerHeight;
          
          // Calculate relative percentage through wrapper viewport scroll
          const totalHeight = wrapperRect.height;
          const currentScroll = windowHeight / 2 - wrapperRect.top;
          
          let progressPercentage = (currentScroll / totalHeight) * 100;
          progressPercentage = Math.max(0, Math.min(100, progressPercentage));
          
          progressLine.style.height = progressPercentage + '%';
        }

        // Throttle updates via RequestAnimationFrame
        let ticking = false;
        window.addEventListener('scroll', function () {
          if (!ticking) {
            window.requestAnimationFrame(function () {
              updateTimelineProgress();
              ticking = false;
            });
            ticking = true;
          }
        });
        updateTimelineProgress();

        /* ------------------------------------------------------------------
           3. Accordion Interactive Toggle (Expand/Collapse)
           ------------------------------------------------------------------ */
        const expandButtons = timelineWrapper.querySelectorAll('.expand-btn');

        expandButtons.forEach(button => {
          button.addEventListener('click', function () {
            const card = this.closest('.timeline-card');
            const isExpanded = card.classList.contains('is-expanded');
            const btnText = this.querySelector('.btn-text');

            if (isExpanded) {
              card.classList.remove('is-expanded');
              this.setAttribute('aria-expanded', 'false');
              if (btnText) btnText.textContent = 'View details';
            } else {
              card.classList.add('is-expanded');
              this.setAttribute('aria-expanded', 'true');
              if (btnText) btnText.textContent = 'Hide details';
            }
          });
        });
      });
    })();
  </script>
