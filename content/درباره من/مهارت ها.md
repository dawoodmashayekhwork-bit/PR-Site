  .skills-radar-wrap{
    background: transparent;   /* بدون پس‌زمینه آبی */
    display:flex;
    justify-content:center;
    padding: 20px 0;
  }
  #skillsRadar text{
    font-family: Tahoma, "Vazirmatn", sans-serif;
    font-size: 14px;
    fill: #1a1a2e;
  }
</style>

<script>
(function(){
  // ---------- داده‌های مهارت‌ها را از اینجا تغییر بده (بین 0 تا 100) ----------
  const skills = [
    { label: "Identity",   value: 85 },
    { label: "Ai",         value: 90 },
    { label: "Typography", value: 95 },
    { label: "Social",     value: 60 },
    { label: "Branding",   value: 90 },
    { label: "Color",      value: 95 },
  ];
  // -----------------------------------------------------------------------

  const svg = document.getElementById('skillsRadar');
  const cx = 200, cy = 190;      // مرکز چارت
  const maxR = 140;              // شعاع بیرونی‌ترین شش‌ضلعی
  const levels = 4;              // تعداد شش‌ضلعی‌های گرید
  const n = skills.length;
  const angleStep = (Math.PI * 2) / n;

  // زاویه هر محور را طوری تنظیم می‌کنیم که راس بالا باشد (مثل عکس نمونه)
  function pointFor(index, radius){
    const angle = -Math.PI/2 + index * angleStep;
    return {
      x: cx + radius * Math.cos(angle),
      y: cy + radius * Math.sin(angle)
    };
  }

  function hexPath(radius){
    let d = "";
    for(let i=0;i<n;i++){
      const p = pointFor(i, radius);
      d += (i===0 ? "M" : "L") + p.x.toFixed(2) + "," + p.y.toFixed(2) + " ";
    }
    return d + "Z";
  }

  let svgContent = "";

  // 1) گرید شش‌ضلعی‌های تودرتو
  for(let l=1; l<=levels; l++){
    const r = (maxR / levels) * l;
    svgContent += `<path d="${hexPath(r)}" fill="none" stroke="#2b2b55" stroke-width="1" opacity="0.6"/>`;
  }

  // 2) خطوط شعاعی از مرکز به هر راس
  for(let i=0;i<n;i++){
    const p = pointFor(i, maxR);
    svgContent += `<line x1="${cx}" y1="${cy}" x2="${p.x.toFixed(2)}" y2="${p.y.toFixed(2)}" stroke="#2b2b55" stroke-width="1" opacity="0.6"/>`;
  }

  // 3) چندضلعی مقدار مهارت‌ها (پر شده)
  let dataPath = "";
  skills.forEach((s, i) => {
    const r = (s.value/100) * maxR;
    const p = pointFor(i, r);
    dataPath += (i===0 ? "M" : "L") + p.x.toFixed(2) + "," + p.y.toFixed(2) + " ";
  });
  dataPath += "Z";

  svgContent += `<path d="${dataPath}" fill="#5b6fd8" fill-opacity="0.55" stroke="#3d4fb5" stroke-width="2"/>`;

  // 4) نقطه روی هر راس داده
  skills.forEach((s, i) => {
    const r = (s.value/100) * maxR;
    const p = pointFor(i, r);
    svgContent += `<circle cx="${p.x.toFixed(2)}" cy="${p.y.toFixed(2)}" r="3.5" fill="#3d4fb5"/>`;
  });

  // 5) لیبل‌ها بیرون از شش‌ضلعی
  skills.forEach((s, i) => {
    const p = pointFor(i, maxR + 26);
    let anchor = "middle";
    if (p.x > cx + 5) anchor = "start";
    else if (p.x < cx - 5) anchor = "end";
    svgContent += `<text x="${p.x.toFixed(2)}" y="${p.y.toFixed(2)}" text-anchor="${anchor}" dominant-baseline="middle" font-weight="600">${s.label}</text>`;
  });

  svg.innerHTML = svgContent;
})();
</script>
