<script>

const style = getComputedStyle(document.querySelector(".skills-radar-wrap"));

const GRID = style.getPropertyValue("--radar-grid").trim();
const TEXT = style.getPropertyValue("--radar-text").trim();
const FILL = style.getPropertyValue("--radar-fill").trim();
const STROKE = style.getPropertyValue("--radar-stroke").trim();
const POINT = style.getPropertyValue("--radar-point").trim();

function createRadarChart(svgId, skills){

    const svg = document.getElementById(svgId);
    if(!svg) return;

    const cx = 200;
    const cy = 190;

    const maxR =
        window.innerWidth < 420 ? 95 :
        window.innerWidth < 600 ? 110 :
        window.innerWidth < 900 ? 125 :
        140;

    const levels = 4;
    const n = skills.length;
    const angleStep = (Math.PI * 2) / n;

    function pointFor(index, radius){
        const angle = -Math.PI/2 + index * angleStep;

        return {
            x: cx + radius * Math.cos(angle),
            y: cy + radius * Math.sin(angle)
        };
    }

    function polygonPath(radius){

        let d = "";

        for(let i=0;i<n;i++){

            const p = pointFor(i,radius);

            d += (i===0?"M":"L") + p.x.toFixed(2) + "," + p.y.toFixed(2) + " ";

        }

        return d+"Z";

    }

    let svgContent="";

    // Grid

    for(let l=1;l<=levels;l++){

        const r=(maxR/levels)*l;

        svgContent+=`<path d="${polygonPath(r)}"
        fill="none"
        stroke="${GRID}"
        stroke-width="0.8"
        opacity=".22"/>`;

    }

    // Axis

    for(let i=0;i<n;i++){

        const p=pointFor(i,maxR);

        svgContent+=`<line
        x1="${cx}"
        y1="${cy}"
        x2="${p.x}"
        y2="${p.y}"
        stroke="${GRID}"
        stroke-width=".8"
        opacity=".22"/>`;

    }

    // Polygon

    let dataPath="";

    skills.forEach((s,i)=>{

        const p=pointFor(i,maxR*s.value/100);

        dataPath+=(i===0?"M":"L")+p.x+","+p.y+" ";

    });

    dataPath+="Z";

    svgContent+=`
    <path
    class="radar-fill"
    d="${dataPath}"
    fill="${FILL}"
    fill-opacity=".14"
    stroke="${STROKE}"
    stroke-width="2"/>
    `;

    // Dots

    skills.forEach((s,i)=>{

        const p=pointFor(i,maxR*s.value/100);

        svgContent+=`
        <circle
        class="radar-dot"
        cx="${p.x}"
        cy="${p.y}"
        r="4"
        fill="${POINT}"
        stroke="white"
        stroke-width="2"/>
        `;

    });

    // Labels

    skills.forEach((s,i)=>{

        const offset =
            window.innerWidth<480 ? 18 :
            window.innerWidth<768 ? 24 :
            34;

        const p=pointFor(i,maxR+offset);

        let anchor="middle";

        if(p.x>cx+5) anchor="start";
        else if(p.x<cx-5) anchor="end";

        svgContent+=`
        <text
        class="radar-label"
        x="${p.x}"
        y="${p.y}"
        text-anchor="${anchor}"
        dominant-baseline="middle">

        ${s.label}

        </text>
        `;

    });

    svg.innerHTML=svgContent;

}
