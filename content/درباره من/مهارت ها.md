<script>
(function(){

  const root = document.documentElement;
  const style = getComputedStyle(root);

  const GRID = style.getPropertyValue("--lightgray").trim();
  const TEXT = style.getPropertyValue("--dark").trim();
  const ACCENT = style.getPropertyValue("--secondary").trim();

  const isDark = document.documentElement.getAttribute("saved-theme") === "dark";

  // ---------- داده‌های مهارت‌ها را از اینجا تغییر بده (بین 0 تا 100) ----------
  const skills = [
    { label: "افترافکت", value: 85 },
    { label: "Ai",        value: 90 },
    { label: "فتوشاپ",   value: 95 },
    { label: "AUTOCAD",   value: 60 },
    { label: "2",         value: 90 },
    { label: "1",         value: 95 },
  ];
  // -----------------------------------------------------------------------

  const svg = document.getElementById('skillsRadar');
  const cx = 200, cy = 190;      // مرکز چارت
  const maxR =
    window.innerWidth < 420 ? 95 :
    window.innerWidth < 600 ? 110 :
    window.innerWidth < 900 ? 125 :
    140;                          // شعاع بیرونی‌ترین شش‌ضلعی
  const levels = 4;              // تعداد شش‌ضلعی‌های گرید
  const n = skills.length;
  const angleStep = (Math.PI * 2) / n;

  // زاویه هر محور را طوری تنظیم می‌کنیم که راس بالا باشد
  function pointFor(index, radius){
    const angle = -Math.PI/2 + index * angleStep;
    return {
      x: cx + radius * Math.cos(angle),
      y: cy + radius * Math.sin(angle)
    };
  }

  function hexPath(radius){
    let d = "";
    for(let i=0;i<n;i++){
      const p = pointFor(i, radius);
      d += (i===0 ? "M" : "L") + p.x.toFixed(2) + "," + p.y.toFixed(2) + " ";
    }
    return d + "Z";
  }

  let svgContent = "";

  // ========= Gradient + Glow =========
  svgContent += `
  <defs>
    <radialGradient id="radarGradient">
      <stop offset="0%" stop-color="${ACCENT}" stop-opacity=".45"/>
      <stop offset="70%" stop-color="${ACCENT}" stop-opacity=".18"/>
      <stop offset="100%" stop-color="${ACCENT}" stop-opacity="0"/>
    </radialGradient>
    <filter id="radarGlow">
      <feDropShadow
        dx="0"
        dy="0"
        stdDeviation="${isDark ? 8 : 5}"
        flood-color="${ACCENT}"
        flood-opacity="${isDark ? .50 : .25}" />
    </filter>
  </defs>
  `;

  // 1) گرید شش‌ضلعی‌های تودرتو
  for(let l=1; l<=levels; l++){
    const r = (maxR / levels) * l;
    svgContent += `<path d="${hexPath(r)}" fill="none" stroke="${GRID}" stroke-width="1" opacity="0.6"/>`;
  }

  // 2) خطوط شعاعی از مرکز به هر راس
  for(let i=0;i<n;i++){
    const p = pointFor(i, maxR);
    svgContent += `<line x1="${cx}" y1="${cy}" x2="${p.x.toFixed(2)}" y2="${p.y.toFixed(2)}" stroke="${GRID}" stroke-width="1" opacity="0.6"/>`;
  }

  // 3) چندضلعی مقدار مهارت‌ها (پر شده)
  let dataPath = "";
  skills.forEach((s, i) => {
    const r = (s.value/100) * maxR;
    const p = pointFor(i, r);
    dataPath += (i===0 ? "M" : "L") + p.x.toFixed(2) + "," + p.y.toFixed(2) + " ";
  });
  dataPath += "Z";

  svgContent += `
  <path
    d="${dataPath}"
    fill="url(#radarGradient)"
    stroke="${ACCENT}"
    stroke-width="2.5"
    filter="url(#radarGlow)"
  />`;

  // 4) نقطه روی هر راس داده
  skills.forEach((s, i) => {
    const r = (s.value/100) * maxR;
    const p = pointFor(i, r);
    svgContent += `<circle cx="${p.x.toFixed(2)}" cy="${p.y.toFixed(2)}" r="3.5" fill="white" stroke="${ACCENT}" stroke-width="2.5"/>`;
  });

  // 5) لیبل‌ها بیرون از شش‌ضلعی
  skills.forEach((s, i) => {
    const labelOffset =
      window.innerWidth < 480 ? 15 :
      window.innerWidth < 768 ? 20 :
      26;
    const p = pointFor(i, maxR + labelOffset);
    let anchor = "middle";
    if (p.x > cx + 5) anchor = "start";
    else if (p.x < cx - 5) anchor = "end";
    svgContent += `<text x="${p.x.toFixed(2)}" y="${p.y.toFixed(2)}" text-anchor="${anchor}" dominant-baseline="middle" font-weight="600">${s.label}</text>`;
  });

  svg.innerHTML = svgContent;
})();
</script>
