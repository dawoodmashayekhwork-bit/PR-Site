  <script>
    (function() {
      'use strict';

      function createRadarChart(svgId, skills) {
        const svg = document.getElementById(svgId);
        if (!svg) return;

        const parentWrap = svg.closest(".skills-radar-wrap");
        const style = parentWrap ? getComputedStyle(parentWrap) : getComputedStyle(document.body);

        const GRID = style.getPropertyValue("--radar-grid").trim() || "rgba(128,128,128,0.3)";
        const FILL = style.getPropertyValue("--radar-fill").trim() || "#0071e3";
        const STROKE = style.getPropertyValue("--radar-stroke").trim() || "#0071e3";
        const POINT = style.getPropertyValue("--radar-point").trim() || "#0071e3";

        const cx = 200;
        const cy = 190;

        const maxR =
          window.innerWidth < 420 ? 95 :
          window.innerWidth < 600 ? 110 :
          window.innerWidth < 900 ? 125 : 140;

        const levels = 4;
        const n = skills.length;
        const angleStep = (Math.PI * 2) / n;

        function pointFor(index, radius) {
          const angle = -Math.PI / 2 + index * angleStep;
          return {
            x: cx + radius * Math.cos(angle),
            y: cy + radius * Math.sin(angle)
          };
        }

        function polygonPath(radius) {
          let d = "";
          for (let i = 0; i < n; i++) {
            const p = pointFor(i, radius);
            d += (i === 0 ? "M" : "L") + p.x.toFixed(2) + "," + p.y.toFixed(2) + " ";
          }
          return d + "Z";
        }

        let svgContent = "";

        // Grid
        for (let l = 1; l <= levels; l++) {
          const r = (maxR / levels) * l;
          svgContent += `<path d="${polygonPath(r)}" fill="none" stroke="${GRID}" stroke-width="1" opacity=".40"/>`;
        }

        // Axis
        for (let i = 0; i < n; i++) {
          const p = pointFor(i, maxR);
          svgContent += `<line x1="${cx}" y1="${cy}" x2="${p.x}" y2="${p.y}" stroke="${GRID}" stroke-width="1" opacity=".40"/>`;
        }

        // Data Polygon
        let dataPath = "";
        skills.forEach((s, i) => {
          const p = pointFor(i, maxR * s.value / 100);
          dataPath += (i === 0 ? "M" : "L") + p.x + "," + p.y + " ";
        });
        dataPath += "Z";

        svgContent += `<path class="radar-fill" d="${dataPath}" fill="${FILL}" fill-opacity=".20" stroke="${STROKE}" stroke-width="2.4"/>`;

        // Dots
        skills.forEach((s, i) => {
          const p = pointFor(i, maxR * s.value / 100);
          svgContent += `<circle class="radar-dot" data-index="${i}" cx="${p.x}" cy="${p.y}" r="4" fill="${POINT}" stroke="white" stroke-width="2"/>`;
        });

        // Labels
        skills.forEach((s, i) => {
          let offset = window.innerWidth < 480 ? 18 : window.innerWidth < 768 ? 24 : 34;

          if (i === 1) offset += 18;
          if (i === 2) offset += 8;

          const p = pointFor(i, maxR + offset);
          let anchor = "middle";

          if (p.x > cx + 5) anchor = "start";
          else if (p.x < cx - 5) anchor = "end";

          svgContent += `<text class="radar-label" data-index="${i}" x="${p.x}" y="${p.y}" text-anchor="${anchor}" dominant-baseline="middle">${s.label}</text>`;
        });

        svg.innerHTML = svgContent;

        // Hover Handlers
        const dots = svg.querySelectorAll(".radar-dot");
        const labels = svg.querySelectorAll(".radar-label");

        dots.forEach(dot => {
          dot.addEventListener("mouseenter", () => {
            const i = dot.dataset.index;
            dot.classList.add("active");
            svg.querySelector(`.radar-label[data-index="${i}"]`)?.classList.add("active");
          });
          dot.addEventListener("mouseleave", () => {
            const i = dot.dataset.index;
            dot.classList.remove("active");
            svg.querySelector(`.radar-label[data-index="${i}"]`)?.classList.remove("active");
          });
        });

        labels.forEach(label => {
          label.addEventListener("mouseenter", () => {
            const i = label.dataset.index;
            label.classList.add("active");
            svg.querySelector(`.radar-dot[data-index="${i}"]`)?.classList.add("active");
          });
          label.addEventListener("mouseleave", () => {
            const i = label.dataset.index;
            label.classList.remove("active");
            svg.querySelector(`.radar-dot[data-index="${i}"]`)?.classList.remove("active");
          });
        });
      }

      function initSkills() {
        // Fallback for card reveal
        setTimeout(() => {
          document.querySelectorAll(".skill-card").forEach(card => card.classList.add("show"));
        }, 600);

        // Architecture
        createRadarChart("skillsRadar", [
          { label: "AutoCAD", value: 85 },
          { label: "SketchUp", value: 88 },
          { label: "Lumion", value: 89 },
          { label: "Revit", value: 60 },
          { label: "3ds Max", value: 40 },
          { label: "Corona", value: 60 },
          { label: "Enscape", value: 85 }
        ]);

        // Creative
        createRadarChart("skillsRadarCreative", [
          { label: "After Effects", value: 95 },
          { label: "Photoshop", value: 95 },
          { label: "InDesign", value: 85 },
          { label: "Illustrator", value: 70 },
          { label: "Audition", value: 70 }
        ]);

        // Productivity
        createRadarChart("skillsRadarProductivity", [
          { label: "MS Project", value: 90 },
          { label: "EndNote", value: 90 },
          { label: "SPSS", value: 80 },
          { label: "Office", value: 80 },
          { label: "Obsidian", value: 70 },
          { label: "VS Code", value: 30 }
        ]);

        // Scroll Observer
        const observer = new IntersectionObserver(entries => {
          entries.forEach(entry => {
            if (entry.isIntersecting) {
              entry.target.classList.add("show");
            }
          });
        }, { threshold: 0.1 });

        document.querySelectorAll(".skill-card").forEach(card => observer.observe(card));
      }

      // Support DOM & Quartz SPA Navigation
      if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', initSkills);
      } else {
        initSkills();
      }

      document.addEventListener('nav', initSkills);
    })();
  </script>
